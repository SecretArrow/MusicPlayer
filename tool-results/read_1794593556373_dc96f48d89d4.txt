    60→/** All liked tracks, memoized. */
    61→export function useLikedTracks(): Track[] {
    62→  const allTracks = useAllTracks();
    63→  const likedTrackIds = usePlayerStore((s) => s.likedTrackIds);
    64→  return useMemo(
    65→    () => allTracks.filter((t) => likedTrackIds.has(t.id)),
    66→    [allTracks, likedTrackIds]
    67→  );
    68→}

    70→/** Returns a stable sorter function honouring the library sort setting. */
    71→export function useSortTrackIds() {
    72→  const allTracks = useAllTracks();
    73→  const librarySort = usePlayerStore((s) => s.librarySort);
    74→  return (ids: string[]): string[] => {
    75→    if (librarySort === "custom") return ids;
    76→    const list = ids
    77→      .map((id) => allTracks.find((t) => t.id === id))
    78→      .filter((t): t is NonNullable<typeof t> => Boolean(t));
    79→    const sorted = [...list];
    80→    if (librarySort === "title") sorted.sort((a, b) => a.title.localeCompare(b.title));
    81→    if (librarySort === "artist") sorted.sort((a, b) => a.artist.localeCompare(b.artist));
    82→    if (librarySort === "duration") sorted.sort((a, b) => b.duration - a.duration);
    83→    if (librarySort === "plays") sorted.sort((a, b) => parseInt(b.plays) - parseInt(a.plays));
    84→    return sorted.map((t) => t.id);
    85→  };
    86→}

   378→function SearchView() {
   379→  const [q, setQ] = useState("");
   380→  const [tab, setTab] = useState<"songs" | "artists" | "albums" | "playlists">("songs");
   381→  const playlists = usePlayerStore((s) => s.playlists);
   382→  const artists = usePlayerStore((s) => s.artists);
   383→  const albums = usePlayerStore((s) => s.albums);
   384→  const setActiveArtist = usePlayerStore((s) => s.setActiveArtist);
   385→  const setActiveAlbum = usePlayerStore((s) => s.setActiveAlbum);
   386→  const setActivePlaylist = usePlayerStore((s) => s.setActivePlaylist);
   387→  const setActiveView = usePlayerStore((s) => s.setActiveView);
   388→  const pushSearchHistory = usePlayerStore((s) => s.pushSearchHistory);
   389→  const searchHistory = usePlayerStore((s) => s.searchHistory);
   390→  const allTracks = useAllTracks();
   391→
   392→  const lower = q.toLowerCase().trim();
   393→  const songs = useMemo(
   394→    () =>
   395→      !lower
   396→        ? []
   397→        : allTracks.filter(
   398→            (t) =>
   399→              t.title.toLowerCase().includes(lower) ||
   400→              t.artist.toLowerCase().includes(lower) ||
   401→              t.album.toLowerCase().includes(lower) ||
   402→              t.genre.toLowerCase().includes(lower)
   403→          ),
   404→    [lower, allTracks]
   405→  );
   406→  const matchedArtists = useMemo(
   407→    () => (!lower ? artists : artists.filter((a) => a.name.toLowerCase().includes(lower))),
   408→    [lower, artists]
   409→  );
   410→  const matchedAlbums = useMemo(
   411→    () => (!lower ? albums : albums.filter((al) => al.title.toLowerCase().includes(lower))),
   412→    [lower, albums]
   413→  );
   414→  const matchedPlaylists = useMemo(
   415→    () => (!lower ? playlists : playlists.filter((p) => p.title.toLowerCase().includes(lower))),
   416→    [lower, playlists]
   417→  );
   418→
   419→  return (
   420→    <div className="space-y-6">
   421→      <div>
   422→        <h1 className="text-2xl font-bold text-white">Search</h1>
   423→        <div className="relative mt-4 max-w-xl">

   426→            autoFocus
   427→            placeholder="Type a song, artist, album or genre…"

   431→            className="rounded-full border-white/10 bg-white/5 pl-9 text-zinc-100 placeholder:text-zinc-500"
   432→          />
   433→        </div>

   437→            <button
   438→              key={t}
   439→              onClick={() => setTab(t)}
   440→              className={cn(
   441→                "rounded-full px-3 py-1 text-xs font-semibold capitalize transition-colors",
   442→                tab === t

   691→function PlaylistView({
   692→  title,
   693→  description,
   694→  cover,
   695→  trackIds,
   696→  editable,
   697→}: {
   698→  title: string;
   699→  description: string;
   700→  cover: string;
   701→  trackIds: string[];
   702→  editable?: boolean;
   703→}) {
   704→  const activePlaylistId = usePlayerStore((s) => s.activePlaylistId);
   705→  const deletePlaylist = usePlayerStore((s) => s.deletePlaylist);
   706→  const setActivePlaylist = usePlayerStore((s) => s.setActivePlaylist);
   707→  const playTrack = usePlayerStore((s) => s.playTrack);
   708→  const allTracks = useAllTracks();
   709→  const list = trackIds
   710→    .map((id) => allTracks.find((t) => t.id === id))
   711→    .filter((t): t is NonNullable<typeof t> => Boolean(t));
   712→  const totalSec = list.reduce((s, t) => s + t.duration, 0);
   713→  return (
   714→    <div className="space-y-6">
   715→      <div className="flex items-end gap-5">
   716→        <div className="h-44 w-44 shrink-0 rounded-lg shadow-2xl" style={{ background: cover }} />
   717→        <div className="min-w-0 pb-2">
   718→          <div className="text-xs font-semibold uppercase tracking-wider text-zinc-400">
   719→            Playlist
   720→          </div>
   721→          <h1 className="mt-1 text-4xl font-bold text-white sm:text-5xl">{title}</h1>
   722→          <p className="mt-2 max-w-xl text-sm text-zinc-400">{description}</p>
   723→          <div className="mt-2 text-xs text-zinc-500">
   724→            {list.length} songs · {Math.floor(totalSec / 60)} min {totalSec % 60} sec
   725→          </div>
   726→        </div>
   727→      </div>
   728→
   729→      <div className="flex items-center gap-4">
   730→        <button
   731→          onClick={() => {
   732→            if (list[0]) playTrack(list[0].id, list.map((t) => t.id));
   733→          }}
   734→          className="flex h-12 w-12 items-center justify-center rounded-full bg-[color:var(--accent)] text-zinc-950 shadow-lg transition-transform hover:scale-105"
   735→        >
   736→          <Play className="ml-0.5 h-5 w-5" fill="currentColor" />
   737→        </button>
   738→        <Button variant="ghost" size="icon" className="text-zinc-400 hover:text-white">
   739→          <Heart className="h-6 w-6" />
   740→        </Button>
   741→        {editable && activePlaylistId && (
   742→          <Button
   743→            variant="ghost"
   744→            size="sm"
   745→            className="ml-auto text-xs text-zinc-400 hover:text-red-400"
   746→            onClick={() => {
   747→              deletePlaylist(activePlaylistId);
   748→              setActivePlaylist(null);
   749→            }}
   750→          >
   751→            Delete playlist
   752→          </Button>
   753→        )}
   754→      </div>
   755→
   756→      {list.length > 0 ? (
   757→        <TrackTable trackIds={trackIds} />
   758→      ) : (
   759→        <div className="rounded-lg border border-dashed border-white/10 p-8 text-center text-sm text-zinc-500">
   760→          This playlist is empty. Use the ⋯ menu on any track row → “Add to playlist” to fill
   761→          it up.
   762→        </div>
   763→      )}
   764→    </div>
   765→  );
   766→}
