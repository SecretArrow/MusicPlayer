    60→    60→/** All liked tracks, memoized. */
    61→    61→export function useLikedTracks(): Track[] {
    62→    62→  const allTracks = useAllTracks();
    63→    63→  const likedTrackIds = usePlayerStore((s) => s.likedTrackIds);
    64→    64→  return useMemo(
    65→    65→    () => allTracks.filter((t) => likedTrackIds.has(t.id)),
    66→    66→    [allTracks, likedTrackIds]
    67→    67→  );
    68→    68→}
    69→    69→
    70→    70→/** Returns a stable sorter function honouring the library sort setting. */
    71→    71→export function useSortTrackIds() {
    72→    72→  const allTracks = useAllTracks();
    73→    73→  const librarySort = usePlayerStore((s) => s.librarySort);
    74→    74→  return (ids: string[]): string[] => {
    75→    75→    if (librarySort === "custom") return ids;
    76→    76→    const list = ids
    77→    77→      .map((id) => allTracks.find((t) => t.id === id))
    78→    78→      .filter((t): t is NonNullable<typeof t> => Boolean(t));
    79→    79→    const sorted = [...list];
    80→    80→    if (librarySort === "title") sorted.sort((a, b) => a.title.localeCompare(b.title));
    81→    81→    if (librarySort === "artist") sorted.sort((a, b) => a.artist.localeCompare(b.artist));
    82→    82→    if (librarySort === "duration") sorted.sort((a, b) => b.duration - a.duration);
    83→    83→    if (librarySort === "plays") sorted.sort((a, b) => parseInt(b.plays) - parseInt(a.plays));
    84→    84→    return sorted.map((t) => t.id);
    85→    85→  };
    86→    86→}

   691→   691→function PlaylistView({
   692→   692→  title,
   693→   693→  description,
   694→   694→  cover,
   695→   695→  trackIds,
   696→   696→  editable,
   697→   697→}: {
   698→   698→  title: string;
   699→   699→  description: string;
   700→   700→  cover: string;
   701→   701→  trackIds: string[];
   702→   702→  editable?: boolean;
   703→   703→}) {
   704→   704→  const activePlaylistId = usePlayerStore((s) => s.activePlaylistId);
   705→   705→  const deletePlaylist = usePlayerStore((s) => s.deletePlaylist);
   706→   706→  const setActivePlaylist = usePlayerStore((s) => s.setActivePlaylist);

   708→   708→  const allTracks = useAllTracks();
   709→   709→  const list = trackIds
   710→   710→    .map((id) => allTracks.find((t) => t.id === id))
   711→   711→    .filter((t): t is NonNullable<typeof t> => Boolean(t));

   713→   713→  return (

   715→   715→      <div className="flex items-end gap-5">

   719→   719→            Playlist
   720→   720→          </div>
   721→   721→          <h1 className="mt-1 text-4xl font-bold text-white sm:text-5xl">{title}</h1>
   722→   722→          <p className="mt-2 max-w-xl text-sm text-zinc-400">{description}</p>
   723→   723→          <div className="mt-2 text-xs text-zinc-500">
   724→   724→            {list.length} songs · {Math.floor(totalSec / 60)} min {totalSec % 60} sec
   725→   725→          </div>
   726→   726→        </div>

   734→   734→          className="flex h-12 w-12 items-center justify-center rounded-full bg-[color:var(--accent)] text-zinc-950 shadow-lg transition-transform hover:scale-105"
