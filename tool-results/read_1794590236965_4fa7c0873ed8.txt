   516→  function SearchView() {
   517→    const [q, setQ] = useState("");
   518→    const [tab, setTab] = useState<"songs" | "artists" | "albums" | "playlists">("songs");
   519→    const pushSearchHistory = usePlayerStore((s) => s.pushSearchHistory);
   520→    const searchHistory = usePlayerStore((s) => s.searchHistory);
   521→
   522→    const lower = q.toLowerCase().trim();
   523→    const songs = useMemo(
   524→      () =>
   525→        !lower
   526→          ? []
   527→          : allTracks.filter(
   528→              (t) =>
   529→                t.title.toLowerCase().includes(lower) ||
   530→                t.artist.toLowerCase().includes(lower) ||
   531→                t.album.toLowerCase().includes(lower) ||
   532→                t.genre.toLowerCase().includes(lower)
   533→            ),
   534→      [lower, allTracks]
   535→    );
   536→    const matchedArtists = useMemo(
   537→      () => (!lower ? artists : artists.filter((a) => a.name.toLowerCase().includes(lower))),
   538→      [lower, artists]
   539→    );
   540→    const matchedAlbums = useMemo(
   541→      () => (!lower ? albums : albums.filter((al) => al.title.toLowerCase().includes(lower))),
   542→      [lower, albums]
   543→    );
   544→    const matchedPlaylists = useMemo(

   808→  function PlaylistView({
   809→    title,
   810→    description,
   811→    cover,
   812→    trackIds,
   813→    editable,
   814→  }: {
   815→    title: string;
   816→    description: string;
   817→    cover: string;
   818→    trackIds: string[];
   819→    editable?: boolean;
   820→  }) {
   821→    const deletePlaylist = usePlayerStore((s) => s.deletePlaylist);
   822→    const list = trackIds
   823→      .map((id) => allTracks.find((t) => t.id === id))
   824→      .filter((t): t is NonNullable<typeof t> => Boolean(t));
   825→    const totalSec = list.reduce((s, t) => s + t.duration, 0);
   826→    return (
   827→      <div className="space-y-6">
   828→        <div className="flex items-end gap-5">
   829→          <div className="h-44 w-44 shrink-0 rounded-lg shadow-2xl" style={{ background: cover }} />
   830→          <div className="min-w-0 pb-2">
   831→            <div className="text-xs font-semibold uppercase tracking-wider text-zinc-400">
   832→              Playlist
   833→            </div>
   834→            <h1 className="mt-1 text-4xl font-bold text-white sm:text-5xl">{title}</h1>
   835→            <p className="mt-2 max-w-xl text-sm text-zinc-400">{description}</p>
   836→            <div className="mt-2 text-xs text-zinc-500">
   837→              {list.length} songs · {Math.floor(totalSec / 60)} min {totalSec % 60} sec
   838→            </div>
   839→          </div>
   840→        </div>
   841→
   842→        <div className="flex items-center gap-4">
   843→          <button
   844→            onClick={() => {
   845→              if (list[0]) playTrack(list[0].id, list.map((t) => t.id));
   846→            }}
   847→            className="flex h-12 w-12 items-center justify-center rounded-full bg-[color:var(--accent)] text-zinc-950 shadow-lg transition-transform hover:scale-105"
   848→          >
   849→            <Play className="ml-0.5 h-5 w-5" fill="currentColor" />
   850→          </button>
   851→          <Button variant="ghost" size="icon" className="text-zinc-400 hover:text-white">
   852→            <Heart className="h-6 w-6" />
   853→          </Button>
   854→          {editable && activePlaylistId && (
   855→            <Button
   856→              variant="ghost"
   857→              size="sm"
   858→              className="ml-auto text-xs text-zinc-400 hover:text-red-400"
   859→              onClick={() => {
   860→                deletePlaylist(activePlaylistId);
   861→                setActivePlaylist(null);
   862→              }}
   863→            >
   864→              Delete playlist
   865→            </Button>
   866→          )}
   867→        </div>
   868→
   869→        {list.length > 0 ? (
   870→          <TrackTable trackIds={trackIds} />
   871→        ) : (
   872→          <div className="rounded-lg border border-dashed border-white/10 p-8 text-center text-sm text-zinc-500">
   873→            This playlist is empty. Use the ⋯ menu on any track row → “Add to playlist” to fill
   874→            it up.
   875→          </div>
   876→        )}
   877→      </div>
   878→    );
   879→  }
