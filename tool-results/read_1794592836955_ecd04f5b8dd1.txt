   527→  function SearchView() {
   528→    const [q, setQ] = useState("");
   529→    const [tab, setTab] = useState<"songs" | "artists" | "albums" | "playlists">("songs");
   530→    const pushSearchHistory = usePlayerStore((s) => s.pushSearchHistory);
   531→    const searchHistory = usePlayerStore((s) => s.searchHistory);
   532→
   533→    const lower = q.toLowerCase().trim();
   534→    const songs = useMemo(
   535→      () =>
   536→        !lower
   537→          ? []
   538→          : allTracks.filter(
   539→              (t) =>
   540→                t.title.toLowerCase().includes(lower) ||
   541→                t.artist.toLowerCase().includes(lower) ||
   542→                t.album.toLowerCase().includes(lower) ||
   543→                t.genre.toLowerCase().includes(lower)
   544→            ),
   545→      [lower, allTracks]

   819→  function PlaylistView({
   820→    title,
   821→    description,
   822→    cover,
   823→    trackIds,
   824→    editable,
   825→  }: {
   826→    title: string;
   827→    description: string;
   828→    cover: string;
   829→    trackIds: string[];
   830→    editable?: boolean;
   831→  }) {
   832→    const deletePlaylist = usePlayerStore((s) => s.deletePlaylist);
   833→    const list = trackIds
   834→      .map((id) => allTracks.find((t) => t.id === id))
   835→      .filter((t): t is NonNullable<typeof t> => Boolean(t));
   836→    const totalSec = list.reduce((s, t) => s + t.duration, 0);
   837→    return (
   838→      <div className="space-y-6">
   839→        <div className="flex items-end gap-5">
   840→          <div className="h-44 w-44 shrink-0 rounded-lg shadow-2xl" style={{ background: cover }} />
   841→          <div className="min-w-0 pb-2">
   842→            <div className="text-xs font-semibold uppercase tracking-wider text-zinc-400">
   843→              Playlist
   844→            </div>
   845→            <h1 className="mt-1 text-4xl font-bold text-white sm:text-5xl">{title}</h1>
   846→            <p className="mt-2 max-w-xl text-sm text-zinc-400">{description}</p>
   847→            <div className="mt-2 text-xs text-zinc-500">
   848→              {list.length} songs · {Math.floor(totalSec / 60)} min {totalSec % 60} sec
   849→            </div>
   850→          </div>
   851→        </div>
   852→
   853→        <div className="flex items-center gap-4">
   854→          <button
   855→            onClick={() => {
   856→              if (list[0]) playTrack(list[0].id, list.map((t) => t.id));

   859→          >

   861→          </button>
   862→          <Button variant="ghost" size="icon" className="text-zinc-400 hover:text-white">
   863→            <Heart className="h-6 w-6" />

   867→              variant="ghost"
   868→              size="sm"
   869→              className="ml-auto text-xs text-zinc-400 hover:text-red-400"
   870→              onClick={() => {

   873→              }}
   874→            >
   875→              Delete playlist
   876→            </Button>

   880→        {list.length > 0 ? (

   885→            it up.
   886→          </div>
   887→        )}
   888→      </div>
