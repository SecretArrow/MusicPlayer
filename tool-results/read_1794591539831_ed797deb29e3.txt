   455→  /** One track row used across home/search/library */
   456→  function TrackRow({ trackId, onActions }: { trackId: string; onActions: () => void }) {
   457→    const t = [...tracks, ...localTracks].find((x) => x.id === trackId);
   458→    if (!t) return null;
   459→    const isActive = queue[currentIndex] === t.id;
   460→    const isLiked = likedTrackIds.has(t.id);
   461→    return (
   462→      <div
   463→        onClick={() => playTrack(t.id, allTracks.map((x) => x.id))}
   464→        className="group flex cursor-pointer items-center gap-3 rounded-md p-1.5 hover:bg-white/[0.05]"
   465→      >
   466→        <div className="relative">
   467→          <CoverArt cover={t.cover} accent={t.accent} title={t.title} size="sm" />
   468→          {isActive && (
   469→            <div className="absolute inset-0 flex items-center justify-center rounded-md bg-black/50">
   470→              {isPlaying ? (
   471→                <Pause className="h-5 w-5 text-[color:var(--accent)]" fill="currentColor" />
   472→              ) : (
   473→                <Play className="ml-0.5 h-5 w-5 text-white" fill="currentColor" />
   474→              )}
   475→            </div>
   476→          )}
   477→        </div>
   478→        <div className="min-w-0 flex-1">
   479→          <div
   480→            className={cn(
   481→              "truncate text-sm font-medium",
   482→              isActive ? "text-[color:var(--accent)]" : "text-white"
   483→            )}
   484→          >
   485→            {t.title}
   486→          </div>
   487→          <div className="truncate text-[11px] text-zinc-400">
   488→            {t.artist} · {t.plays} plays
   489→          </div>
   490→        </div>
   491→        <button
   492→          onClick={(e) => {
   493→            e.stopPropagation();
   494→            toggleLike(t.id);
   495→          }}
   496→          className={cn("p-1", isLiked ? "text-[color:var(--accent)]" : "text-zinc-500")}
   497→          aria-label={isLiked ? "Remove from liked" : "Add to liked"}
   498→        >
   499→          <Heart className={cn("h-4 w-4", isLiked && "fill-current")} />
   500→        </button>
   501→        <button
   502→          onClick={(e) => {
   503→            e.stopPropagation();
   504→            onActions();
   505→          }}
   506→          className="p-1 text-zinc-500"
   507→          aria-label={`Actions for ${t.title}`}
   508→        >
   509→          <MoreHorizontal className="h-4 w-4" />
   510→        </button>
   511→      </div>
   512→    );
   513→  }

   516→function NowPlayingSheet({
   517→  sheet,
   518→  setSheet,
   519→}: {
   520→  sheet: "lyrics" | "queue" | "visual";
   521→  setSheet: (s: Sheet) => void;
   522→}) {
   523→  const track = useCurrentTrack();
   524→  const isPlaying = usePlayerStore((s) => s.isPlaying);
   525→  const togglePlay = usePlayerStore((s) => s.togglePlay);
   526→  const next = usePlayerStore((s) => s.next);
   527→  const prev = usePlayerStore((s) => s.prev);
   528→  const shuffle = usePlayerStore((s) => s.shuffle);
   529→  const toggleShuffle = usePlayerStore((s) => s.toggleShuffle);
   530→  const repeat = usePlayerStore((s) => s.repeat);
   531→  const cycleRepeat = usePlayerStore((s) => s.cycleRepeat);
   532→  const currentTime = usePlayerStore((s) => s.currentTime);
   533→  const duration = usePlayerStore((s) => s.duration);
   534→  const seek = usePlayerStore((s) => s.seek);
   535→  const likedTrackIds = usePlayerStore((s) => s.likedTrackIds);
   536→  const toggleLike = usePlayerStore((s) => s.toggleLike);
   537→  const setActiveArtist = usePlayerStore((s) => s.setActiveArtist);
   538→
   539→  const dur = duration || track?.duration || 0;
   540→  const isLiked = track ? likedTrackIds.has(track.id) : false;
   541→  const RepeatIcon = repeatIcon[repeat];
   542→
   543→  if (!track) return null;
